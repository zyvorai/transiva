🚀 Quick Start:

  # Create a BackupJob for a KubeVirt VM
  cat <<EOF | kubectl apply -f -
  apiVersion: transiva.zyvor.ai/v1alpha1
  kind: BackupJob
  metadata:
    name: my-vm-backup
    namespace: default
  spec:
    source:
      provider: kubevirt
      namespace: default
      vmName: my-vm
    destination:
      type: s3
      bucket: my-backups
      region: us-west-2
{{- if .Values.carbonAware.defaultEnabled }}
    carbonAware:
      enabled: true
      zone: {{ .Values.carbonAware.defaultZone }}
{{- end }}
  EOF

  # Check backup status
  kubectl get backupjobs

  # Create a BackupSchedule for nightly backups
  cat <<EOF | kubectl apply -f -
  apiVersion: transiva.zyvor.ai/v1alpha1
  kind: BackupSchedule
  metadata:
    name: nightly-backup
    namespace: default
  spec:
    schedule: "0 2 * * *"
    timezone: {{ .Values.scheduleDefaults.timezone }}
    jobTemplate:
      spec:
        source:
          provider: kubevirt
          namespace: default
          vmName: my-vm
        destination:
          type: s3
          bucket: my-backups
  EOF
